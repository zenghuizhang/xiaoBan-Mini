# 小伴 Mini V2.0 - 软件架构与实现

> Arduino for ESP32，简单才是王道

---

## 一、技术选型

| 项 | 选择 | 理由 |
|---|------|------|
| 开发框架 | **Arduino for ESP32** | 资料多、上手快、库丰富，适合这种简单项目 |
| 屏幕驱动 | **U8g2 库** | 支持 SSD1306，API 简单 |
| 舵机控制 | **ESP32Servo 库** | ESP32 专用，PWM 稳定 |
| 开发工具 | **Arduino IDE** 或 **VS Code + PlatformIO** | 都可以，看你习惯 |

## 三、核心代码框架

### 1. config.h（配置文件）

```cpp
#ifndef CONFIG_H
#define CONFIG_H

// 引脚定义
#define PIN_SERVO_NOD 2    // 点头舵机
#define PIN_SERVO_TILT 3   // 歪头舵机
#define PIN_TOUCH 4        // 触摸传感器
#define PIN_LIGHT 5        // 光敏电阻（ADC）
#define PIN_SOUND 6        // 咪头（ADC）

// OLED 引脚（I2C）
#define OLED_SDA 1
#define OLED_SCL 0

// 参数
#define TIMER_IDLE_MS 30000      // 30秒进入发呆
#define TIMER_SLEEP_MS 180000    // 3分钟进入睡眠
#define SOUND_THRESHOLD 500       // 声音阈值（ADC值）
#define LIGHT_THRESHOLD_LOW 100   // 天黑阈值
#define LIGHT_THRESHOLD_HIGH 300  // 天亮阈值

// 舵机角度
#define SERVO_NOD_CENTER 90
#define SERVO_NOD_DOWN 70
#define SERVO_NOD_UP 110
#define SERVO_TILT_CENTER 90
#define SERVO_TILT_LEFT 75
#define SERVO_TILT_RIGHT 105

#endif // CONFIG_H
```

### 2. main.ino（主循环）

```cpp
#include <Arduino.h>
#include "config.h"
#include "eyes.h"
#include "servos.h"
#include "sensors.h"
#include "state_machine.h"

void setup() {
  Serial.begin(115200);
  
  eyes_init();    // 初始化眼睛（OLED）
  servos_init();  // 初始化舵机
  sensors_init(); // 初始化传感器
  state_init();   // 初始化状态机
  
  eyes_set_neutral(); // 初始表情：中立
  servos_center();    // 初始位置：居中
}

void loop() {
  // 1. 读取传感器
  bool touch = sensors_read_touch();
  int sound = sensors_read_sound();
  int light = sensors_read_light();
  
  // 2. 更新状态机
  state_update(touch, sound, light);
  
  // 3. 根据状态更新输出
  switch (state_get()) {
    case STATE_SLEEP:
      eyes_off();
      servos_sleep();
      break;
      
    case STATE_AWAKE:
      eyes_set_neutral();
      servos_center();
      break;
      
    case STATE_EXCITED:
      if (touch) {
        eyes_set_happy();
        servos_nod(2); // 点头2次
      } else if (sound > SOUND_THRESHOLD) {
        eyes_set_curious();
        servos_tiny_nod(); // 轻轻点头
      }
      break;
  }
  
  delay(50); // 循环周期 50ms
}
```

```cpp
#ifndef EYES_H
```

```cpp
// eyes.cpp
#include "eyes.h"
#include "config.h"

U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, /* reset=*/ U8X8_PIN_NONE);

void eyes_init() {
  u8g2.begin();
  u8g2.clearBuffer();
}

void eyes_off() {
  u8g2.clearBuffer();
  u8g2.sendBuffer();
}

void draw_eyes(int left_x, int left_y, int right_x, int right_y, const char* mouth) {
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_ncenB14_tr);
  
  // 画眼睛（简单的圆或弯月）
  u8g2.drawCircle(left_x, left_y, 8);
  u8g2.drawCircle(right_x, right_y, 8);
  
  // 画嘴巴（如果有）
  if (mouth) {
    u8g2.drawStr(56, 45, mouth);
  }
  
  u8g2.sendBuffer();
}

void eyes_set_neutral() {
  draw_eyes(40, 25, 88, 25, "-");
}

void eyes_set_happy() {
  // 开心：眼睛弯成月牙（简化版：画横线）
  u8g2.clearBuffer();
  u8g2.drawHLine(32, 25, 16); // 左眼
  u8g2.drawHLine(80, 25, 16); // 右眼
  u8g2.drawStr(56, 45, "~");  // 嘴巴
  u8g2.sendBuffer();
}

void eyes_set_curious() {
  draw_eyes(40, 25, 88, 25, "o");
}

void eyes_set_sleepy() {
  u8g2.clearBuffer();
  u8g2.drawHLine(36, 25, 8); // 左眼（半睁）
  u8g2.drawHLine(84, 25, 8); // 右眼（半睁）
  u8g2.sendBuffer();
}

void eyes_blink() {
  eyes_off();
  delay(100);
  eyes_set_neutral();
}
```

```cpp
// servos.cpp
#include <ESP32Servo.h>
#include "servos.h"
#include "config.h"

Servo servo_nod;
Servo servo_tilt;

void servos_init() {
  servo_nod.attach(PIN_SERVO_NOD);
  servo_tilt.attach(PIN_SERVO_TILT);
  servos_center();
}

void servos_center() {
  servo_nod.write(SERVO_NOD_CENTER);
  servo_tilt.write(SERVO_TILT_CENTER);
}

void servos_sleep() {
  servo_nod.write(SERVO_NOD_DOWN);
  servo_tilt.write(SERVO_TILT_CENTER);
}

void servos_nod(int times) {
  for (int i = 0; i < times; i++) {
    servo_nod.write(SERVO_NOD_DOWN);
    delay(300);
    servo_nod.write(SERVO_NOD_UP);
    delay(300);
  }
  servos_center();
}

void servos_tiny_nod() {
  // 轻轻点头（幅度小）
  servo_nod.write(SERVO_NOD_CENTER - 5);
  delay(200);
  servo_nod.write(SERVO_NOD_CENTER + 5);
  delay(200);
  servos_center();
}

void servos_tilt_left() {
  servo_tilt.write(SERVO_TILT_LEFT);
  delay(500);
  servos_center();
}

void servos_tilt_right() {
  servo_tilt.write(SERVO_TILT_RIGHT);
  delay(500);
  servos_center();
}
```

### 5. state_machine.h / state_machine.cpp（状态机）

```cpp
// state_machine.h
#ifndef STATE_MACHINE_H
#define STATE_MACHINE_H

enum State {
  STATE_SLEEP,
  STATE_AWAKE,
  STATE_EXCITED
};

void state_init();
void state_update(bool touch, int sound, int light);
State state_get();

#endif // STATE_MACHINE_H
```

```cpp
// state_machine.cpp
#include "state_machine.h"
#include "config.h"

static State current_state = STATE_AWAKE;
static unsigned long last_activity_ms = 0;
static unsigned long last_sleep_check_ms = 0;

void state_init() {
  current_state = STATE_AWAKE;
  last_activity_ms = millis();
}

void state_update(bool touch, int sound, int light) {
  unsigned long now = millis();
  
  // 有活动：更新时间
  if (touch || sound > SOUND_THRESHOLD) {
    last_activity_ms = now;
  }
  
  // 状态跳转逻辑
  switch (current_state) {
    case STATE_SLEEP:
      // 睡眠中：有触摸或天亮 → 醒来
      if (touch || light > LIGHT_THRESHOLD_HIGH) {
        current_state = STATE_AWAKE;
        last_activity_ms = now;
      }
      break;
      
    case STATE_AWAKE:
      // 清醒中：有触摸/声音 → 兴奋
      if (touch || sound > SOUND_THRESHOLD) {
        current_state = STATE_EXCITED;
      }
      // 3分钟没活动 + 天黑 → 睡眠
      else if (now - last_activity_ms > TIMER_SLEEP_MS && light < LIGHT_THRESHOLD_LOW) {
        current_state = STATE_SLEEP;
      }
      break;
      
    case STATE_EXCITED:
      // 兴奋中：10秒没活动 → 回到清醒
      if (now - last_activity_ms > 10000) {
        current_state = STATE_AWAKE;
      }
      break;
  }
}

State state_get() {
  return current_state;
}
```

## 四、开发环境搭建

### 1. 安装 Arduino IDE
1. 下载：https://www.arduino.cc/en/software
2. 安装 ESP32 开发板支持：
   - 文件 → 首选项 → 附加开发板管理器网址：
     `https://dl.espressif.com/dl/package_esp32_index.json`
   - 工具 → 开发板 → 开发板管理器 → 搜索 "ESP32" → 安装

### 2. 安装库
- **U8g2**：库管理器 → 搜索 "U8g2" → 安装
- **ESP32Servo**：库管理器 → 搜索 "ESP32Servo" → 安装

### 3. 选择开发板
- 工具 → 开发板 → ESP32 Arduino → "ESP32C3 Dev Module"

## 五、调试技巧

1. **串口调试**：所有状态、传感器值都打印到串口（115200波特率）
2. **单独测试**：
   - 先测舵机：写个简单程序让它动
   - 再测屏幕：写个简单程序显示图案
   - 最后测传感器：读触摸/光线/声音值
3. **延迟要慢**：调试时动作放慢一点，方便看问题
