# xiaoBan-Mini V2.5 - UI 交互设计
> 融合陪伴机器人 UI 理念，打造桌面萌宠的视觉交互

---

## 一、设计理念融合

### 1.1 核心理念
将陪伴机器人 PRD 中的视觉设计理念与 xiaoBan-Mini 的硬件特性结合：

| PRD 理念 | xiaoBan-Mini 适配 |
|---------|------------------|
| 极简界面 | 单屏表情，无菜单栏 |
| 生命感表现 | 随机表情轮播 + 微动作 |
| 昼夜模式 | 光线自适应切换 |
| 视觉重心优化 | 眼睛位置动态调整 |
| 弹性动画 | 舵机平滑过渡动画 |

### 1.2 设计原则
- **如无必要，勿增实体**：保持极简，表情即界面
- **被动而非主动**：不主动打扰，只在互动时响应
- **慢即是快**：动作慢一点，更治愈
- **少即是多**：几个核心表情足够表达情绪

### 3.2 静默状态行为（PRD 核心功能）

在清醒态下，如果 30 秒无操作，进入发呆态：

```cpp
// 发呆态行为（借鉴 PRD 的随机表情轮播）
void idle_behavior() {
  static unsigned long last_expression_change = 0;
  static unsigned long last_blink = 0;
  unsigned long now = millis();
  
  // 1. 随机表情轮播（4-7秒切换，参考 PRD）
  if (now - last_expression_change > random(4000, 7000)) {
    // 随机选择：发呆、可爱、调皮、无聊
    ExpressionState states[] = {EXPR_IDLE, EXPR_CUTE, EXPR_NAUGHTY, EXPR_BORED};
    ExpressionState new_state = states[random(4)];
    eyes_set_expression(new_state);
    last_expression_change = now;
  }
  
  // 2. 随机眨眼（3-5秒，参考 PRD）
  if (now - last_blink > random(3000, 5000)) {
    eyes_blink();
    last_blink = now;
  }
}
```

### 3.3 触摸交互（优化版）

```cpp
void on_touch() {

  servo_tilt.write(TILT_LEFT);
  delay_with_ease(300);  // 弹性动画
  
  // 第二步：眼睛变开心

    delay_with_ease(300);
    servo_nod.write(NOD_UP);
    delay_with_ease(300);
  }
  
  // 第四步：回正
  servo_center();
  eyes_set_expression(EXPR_IDLE);
}
```

## 四、昼夜模式设计（PRD 核心功能）

### 4.1 光线自适应切换

```cpp
// 昼夜模式枚举（参考 PRD）
enum ThemeMode {
  THEME_LIGHT,  // 白天模式
  THEME_DARK    // 黑夜模式
};

// 光线阈值（滞回设计，避免频繁切换）
#define LIGHT_THRESHOLD_DAY    600   // 天亮阈值
#define LIGHT_THRESHOLD_NIGHT   200  // 天黑阈值

// 模式切换逻辑
void check_light_mode() {
  int light = analogRead(PIN_LIGHT);
  
  static ThemeMode current_theme = THEME_LIGHT;
  
  // 白天 → 黑夜
  if (current_theme == THEME_LIGHT && light < LIGHT_THRESHOLD_NIGHT) {
    current_theme = THEME_DARK;
    enter_night_mode();
  }
  // 黑夜 → 白天
  else if (current_theme == THEME_DARK && light > LIGHT_THRESHOLD_DAY) {
    current_theme = THEME_LIGHT;
    enter_day_mode();
  }
}
```

### 4.2 夜间模式行为

```cpp
void enter_night_mode() {
  // 1. 慢慢低头
  servo_nod.write(NOD_DOWN);
  delay_with_ease(1000);
  
  // 2. 眼睛慢慢闭上
  eyes_close_slowly(1000);
  
  // 3. 进入睡眠态（舵机断电，省电）
  servos_detach();
  state = STATE_SLEEP;
}

void enter_day_mode() {
  // 1. 舵机上电
  servos_attach();
  
  // 2. 慢慢抬头
  servo_nod.write(NOD_CENTER);
  delay_with_ease(1000);
  
  // 3. 眼睛慢慢睁开
  eyes_open_slowly(1000);
  
  // 4. 进入清醒态
  state = STATE_AWAKE;
}
```

## 五、动画参数设计（PRD 弹性动画）

### 5.1 弹性动画实现

参考 PRD 的 Spring 动画参数：

```cpp
// 弹性动画参数（PRD: stiffness:300, damping:20）
struct SpringParams {
  float stiffness;   // 刚度（越大越快）
  float damping;     // 阻尼（越大越不弹）
};

// 默认参数
const SpringParams SPRING_DEFAULT = {300, 20};
const SpringParams SPRING_FAST = {400, 25};    // 眨眼
const SpringParams SPRING_SLOW = {280, 22};    // 底部栏

// 弹性动画函数
void delay_with_ease(int target_delay) {
  // 简化版：非线性延迟
  // 实际可以用更复杂的弹簧算法
  int steps = 10;
  for (int i = 0; i < steps; i++) {
    float progress = (float)i / steps;
    // 模拟弹簧曲线
    float eased = 1 - pow(1 - progress, 3);
    delay(target_delay / steps);
  }
}
```

### 5.2 表情切换动画

```cpp
// 表情切换（平滑过渡）
void eyes_set_expression(ExpressionState new_expr) {
  static ExpressionState current_expr = EXPR_IDLE;
  
  if (new_expr == current_expr) return;
  
  // 弹性过渡动画（参考 PRD: 400ms）
  unsigned long start_time = millis();
  const int transition_duration = 400;
  
  while (millis() - start_time < transition_duration) {
    float progress = (millis() - start_time) / (float)transition_duration;
    // 插值计算中间状态
    draw_eyes_interpolated(current_expr, new_expr, progress);
  }
  
  current_expr = new_expr;
  draw_eyes_final(new_expr);
}
```

```cpp
#include <U8g2lib.h>

U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, U8X8_PIN_NONE);

void draw_eye(int x, int y, int width, int height, int radius, bool filled = true) {
  if (filled) {
    u8g2.drawFilledEllipse(x, y, width/2, height/2);
  } else {
    u8g2.drawEllipse(x, y, width/2, height/2);

// 绘制弯月形眼睛（开心状态）
void draw_happy_eye(int x, int y, int width, int height) {
  // 上弧线
  u8g2.drawArc(x - width/2, y, width/2, height/2, 180, 360);
  // 下弧线（微笑）
  u8g2.drawArc(x, y + height/4, width/4, height/4, 0, 180);

// 绘制完整表情
void draw_expression(ExpressionState state) {
  u8g2.clearBuffer();
  
  int left_x = 40;
  int right_x = 88;
  int y = EYE_POSITIONS[state];
  
  switch (state) {

      draw_happy_eye(right_x, y, 16, 8);
```

## 七、状态机完整实现

### 7.1 状态定义

```cpp
enum RobotState {
  STATE_SLEEP,    // 睡眠态
  STATE_AWAKE,    // 清醒态
  STATE_IDLE,     // 发呆态
  STATE_EXCITED   // 兴奋态
};

RobotState current_state = STATE_AWAKE;
unsigned long last_activity_time = 0;
```

### 7.2 主循环

```cpp
void loop() {
  unsigned long now = millis();
  
  // 1. 读取传感器
  bool touch = digitalRead(PIN_TOUCH) == HIGH;
  int sound = analogRead(PIN_SOUND);
  int light = analogRead(PIN_LIGHT);
  
  // 2. 检查昼夜模式
  check_light_mode();
  
  // 3. 状态机处理
  switch (current_state) {
    case STATE_SLEEP:
      // 睡眠中：检测触摸或天亮
      if (touch || light > LIGHT_THRESHOLD_DAY) {
        enter_day_mode();
        current_state = STATE_AWAKE;
        last_activity_time = now;
      }
      break;
      
    case STATE_AWAKE:
      // 清醒态：检测交互
      if (touch) {
        on_touch();
        current_state = STATE_EXCITED;
        last_activity_time = now;
      }
      else if (sound > SOUND_THRESHOLD) {
        on_sound_detected();
        last_activity_time = now;
      }
      // 30秒无操作 → 发呆态
      else if (now - last_activity_time > 30000) {
        current_state = STATE_IDLE;
      }
      // 天黑 → 睡眠
      else if (light < LIGHT_THRESHOLD_NIGHT) {
        enter_night_mode();
        current_state = STATE_SLEEP;
      }
      break;
      
    case STATE_IDLE:
      // 发呆态：随机表情轮播
      idle_behavior();
      
      // 有交互 → 兴奋态
      if (touch || sound > SOUND_THRESHOLD) {
        current_state = STATE_EXCITED;
        last_activity_time = now;
      }
      // 3分钟无操作 + 天黑 → 睡眠
      else if (now - last_activity_time > 180000 && light < LIGHT_THRESHOLD_NIGHT) {
        enter_night_mode();
        current_state = STATE_SLEEP;
      }
      break;
      
    case STATE_EXCITED:
      // 兴奋态：动作完成后回到清醒态
      if (now - last_activity_time > 10000) {
        eyes_set_expression(EXPR_IDLE);
        servo_center();
        current_state = STATE_AWAKE;
      }
      break;
  }
  
  delay(50);  // 20Hz 循环
}
```
